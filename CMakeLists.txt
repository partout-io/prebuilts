cmake_minimum_required(VERSION 3.28)
project(Prebuilts LANGUAGES C)

include(ExternalProject)
include(FetchContent)
include(GNUInstallDirs)

option(PPV_BUILD_OPENSSL "Build OpenSSL" OFF)
option(PPV_BUILD_MBEDTLS "Build Mbed TLS" OFF)
option(PPV_BUILD_WG_GO "Build wg-go" OFF)
option(PPV_BUILD_WINTUN "Build Wintun on Windows" OFF)
set(PPV_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/output" CACHE PATH
    "Vendor staging directory")
file(TO_CMAKE_PATH "${PPV_OUTPUT}" PPV_OUTPUT)

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" PPV_ARCH)
if(WIN32)
    if(CMAKE_C_COMPILER_ARCHITECTURE_ID)
        string(TOLOWER "${CMAKE_C_COMPILER_ARCHITECTURE_ID}" PPV_ARCH)
    elseif(DEFINED ENV{VSCMD_ARG_TGT_ARCH})
        string(TOLOWER "$ENV{VSCMD_ARG_TGT_ARCH}" PPV_ARCH)
    endif()
    if(PPV_ARCH MATCHES "^(x64|x86_64|amd64)(-|$)")
        set(PPV_ARCH amd64)
    elseif(PPV_ARCH MATCHES "^(arm64|aarch64)(-|$)")
        set(PPV_ARCH arm64)
    endif()
endif()

if(APPLE AND CMAKE_SYSTEM_NAME STREQUAL "Darwin")
    if(NOT CMAKE_OSX_SYSROOT)
        execute_process(
            COMMAND xcrun --sdk macosx --show-sdk-path
            OUTPUT_VARIABLE CMAKE_OSX_SYSROOT
            OUTPUT_STRIP_TRAILING_WHITESPACE
            COMMAND_ERROR_IS_FATAL ANY
        )
    endif()
    if(NOT CMAKE_C_COMPILER_TARGET)
        set(CMAKE_C_COMPILER_TARGET "${PPV_ARCH}-apple-macos")
    endif()
endif()

if(WIN32)
    set(PPV_MAKE nmake)
else()
    set(PPV_MAKE make)
endif()
if(CMAKE_CONFIGURATION_TYPES)
    set(PPV_CONFIG_ARGS --config $<CONFIG>)
endif()

set(VENDOR_ENV)
if(ANDROID)
    file(GLOB android_toolchains "${CMAKE_ANDROID_NDK}/toolchains/llvm/prebuilt/*")
    list(GET android_toolchains 0 android_toolchain)
    if(CMAKE_HOST_WIN32)
        set(vendor_path "${android_toolchain}/bin;$ENV{PATH}")
        string(REPLACE ";" "\\;" vendor_path "${vendor_path}")
    else()
        set(vendor_path "${android_toolchain}/bin:$ENV{PATH}")
    endif()
    set(VENDOR_ENV "${CMAKE_COMMAND}" -E env "PATH=${vendor_path}")
endif()

set(PPV_VENDOR_DEPENDENCIES)
if(PPV_BUILD_OPENSSL)
    include("cmake/vendors/openssl.cmake")
    list(APPEND PPV_VENDOR_DEPENDENCIES OpenSSLProject)
    install(DIRECTORY "${OPENSSL_DIR}/" DESTINATION openssl USE_SOURCE_PERMISSIONS)
endif()
if(PPV_BUILD_MBEDTLS)
    include("cmake/vendors/mbedtls.cmake")
    list(APPEND PPV_VENDOR_DEPENDENCIES MbedTLSProject)
    install(DIRECTORY "${MBEDTLS_DIR}/" DESTINATION mbedtls USE_SOURCE_PERMISSIONS)
endif()
if(PPV_BUILD_WG_GO)
    include("cmake/vendors/wg-go.cmake")
    list(APPEND PPV_VENDOR_DEPENDENCIES WireGuardGoProject)
    install(DIRECTORY "${WGGO_DIR}/" DESTINATION wg-go USE_SOURCE_PERMISSIONS)
endif()
if(PPV_BUILD_WINTUN)
    if(NOT WIN32)
        message(FATAL_ERROR "Wintun can only be built on Windows")
    endif()
    include("cmake/vendors/wintun.cmake")
    list(APPEND PPV_VENDOR_DEPENDENCIES WintunProject)
    install(DIRECTORY "${WINTUN_DIR}/" DESTINATION wintun USE_SOURCE_PERMISSIONS)
endif()

add_custom_target(vendors ALL DEPENDS ${PPV_VENDOR_DEPENDENCIES})
